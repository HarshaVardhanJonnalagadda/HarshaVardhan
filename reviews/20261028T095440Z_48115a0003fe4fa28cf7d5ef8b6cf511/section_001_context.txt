0013: def read_source(filename: str) -> str:
0014:     path = Path(filename)
0015: 
0016:     if path.suffix.lower() != ".py":
0017:         raise ValueError("Please provide a Python (.py) file.")
0018: 
0019:     try:
0020:         source = path.read_text(encoding="utf-8")
0021:     except FileNotFoundError:
0022:         raise ValueError(f"File not found: {path}") from None
0023:     except UnicodeDecodeError:
0024:         raise ValueError("The source must be valid UTF-8 text.") from None
0025:     except OSError as exc:
0026:         raise ValueError(f"Cannot read the source: {exc}") from None
0027: 
0028:     if not source.strip():
0029:         raise ValueError("The source file is empty.")
0030: 
0031:     if "\x00" in source:
0032:         raise ValueError("The source contains invalid null characters.")
0033: 
0034:     return source

0080: def split_source(source: str, max_bytes: int = 8000) -> list[str]:
0081:     """Split source into sections while retaining original line numbers."""
0082:     sections = []
0083:     current_lines = []
0084:     current_size = 0
0085: 
0086:     for number, line in enumerate(source.splitlines(), start=1):
0087:         numbered_line = f"{number}: {line}"
0088:         line_size = len((numbered_line + "\n").encode("utf-8"))
0089: 
0090:         if line_size > max_bytes:
0091:             raise ValueError(
0092:                 f"Source line {number} is too long for one review section."
0093:             )
0094: 
0095:         if current_lines and current_size + line_size > max_bytes:
0096:             sections.append("\n".join(current_lines))
0097:             current_lines = []
0098:             current_size = 0
0099: 
0100:         current_lines.append(numbered_line)
0101:         current_size += line_size
0102: 
0103:     if current_lines:
0104:         sections.append("\n".join(current_lines))
0105: 
0106:     return sections
