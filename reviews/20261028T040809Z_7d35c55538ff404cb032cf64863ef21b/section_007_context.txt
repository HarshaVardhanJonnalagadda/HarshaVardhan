0857: def import_orders(db: Database, path: Path, now: datetime) -> dict:
0858:     if path.stat().st_size > 1_000_000:
0859:         raise ServiceError("Import exceeds 1000000 bytes")
0860:     service = OrderService(db)
0861:     accepted = []
0862:     rejected = []
0863:     with path.open(encoding="utf-8") as stream:
0864:         for number, line in enumerate(stream, 1):
0865:             if not line.strip():
0866:                 continue
0867:             if number > 1000:
0868:                 rejected.append({"line": number, "error": "Import line limit reached"})
0869:                 break
0870:             try:
0871:                 value = json.loads(line)
0872:                 if not isinstance(value, dict):
0873:                     raise ServiceError("Each line must be an object")
0874:                 request = CheckoutRequest.from_dict(value)
0875:                 order = service.checkout(request, now)
0876:                 accepted.append({"line": number, "order_id": order["order_id"]})
0877:             except (ServiceError, json.JSONDecodeError) as exc:
0878:                 rejected.append({"line": number, "error": str(exc)})
0879:     return {"accepted": accepted, "rejected": rejected}
