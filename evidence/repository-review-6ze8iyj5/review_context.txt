38: def parse_response(raw):
39:     try:
40:         data = json.loads(raw)
41:         choice = data["choices"][0]
42:         content = choice["message"]["content"]
43:     except (ValueError, KeyError, IndexError, TypeError):
44:         raise ProviderError(
45:             "Provider returned a malformed response.",
46:             raw_response=raw,
47:         ) from None
48: 
49:     if not isinstance(content, str) or not content.strip():
50:         raise ProviderError(
51:             "Provider returned an empty response.",
52:             raw_response=raw,
53:         )
54: 
55:     if choice.get("finish_reason") != "stop":
56:         raise ProviderError(
57:             "Response was incomplete or its completion status was unknown.",
58:             raw_response=raw,
59:         )
60: 
61:     usage = data.get("usage")
62:     if not isinstance(usage, dict):
63:         usage = {}
64: 
65:     def token_count(name):
66:         value = usage.get(name)
67:         if type(value) is int and value >= 0:
68:             return value
69:         return "unknown"
70: 
71:     return {
72:         "text": content,
73:         "raw_response": raw,
74:         "usage": {
75:             "input_tokens": token_count("prompt_tokens"),
76:             "output_tokens": token_count("completion_tokens"),
77:             "total_tokens": token_count("total_tokens"),
78:         },
79:     }
